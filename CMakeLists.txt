cmake_minimum_required(VERSION 3.25)
if(CMAKE_HOST_APPLE AND (NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET OR CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL ""))
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version" FORCE)
endif()
project(reVoice VERSION 0.3.0 DESCRIPTION "Real-time vocal harmonizer" LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(REVOICE_BUILD_APP "Build the native FLTK desktop application" ON)
option(REVOICE_BUILD_RENDERER "Build the offline WAV renderer" ON)
option(REVOICE_BUILD_PLUGINS "Build VST3 and optional AU plug-ins" OFF)
option(REVOICE_BUILD_AU "Build the macOS Audio Unit plug-in" ${APPLE})
option(REVOICE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(REVOICE_BUILD_PLUGINS)
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  set(CMAKE_CXX_VISIBILITY_PRESET hidden)
  set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
  if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
include(CTest)
include(GNUInstallDirs)
if(APPLE AND REVOICE_BUILD_APP)
  set(REVOICE_DATA_DIR "reVoice.app/Contents/Resources")
else()
  set(REVOICE_DATA_DIR "${CMAKE_INSTALL_DATADIR}/reVoice")
endif()

add_library(revoice_options INTERFACE)
target_compile_features(revoice_options INTERFACE cxx_std_17)
if(MSVC)
  target_compile_definitions(revoice_options INTERFACE NOMINMAX _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS)
  target_compile_options(revoice_options INTERFACE /W4 /permissive-)
else()
  target_compile_options(revoice_options INTERFACE -Wall -Wextra -Wno-unused-parameter)
  if(REVOICE_SANITIZERS)
    target_compile_options(revoice_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(revoice_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()

include(cmake/Dependencies.cmake)
add_subdirectory(libs/pandora)
add_subdirectory(engine)
if(REVOICE_BUILD_RENDERER)
  add_executable(revoice-render app/render.cpp)
  target_compile_definitions(revoice-render PRIVATE REVOICE_VERSION="${PROJECT_VERSION}")
  target_link_libraries(revoice-render PRIVATE revoice_engine revoice_options)
  install(TARGETS revoice-render RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT Runtime)
endif()
if(REVOICE_BUILD_APP)
  add_subdirectory(app)
endif()
if(REVOICE_BUILD_PLUGINS)
  add_subdirectory(plugins)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()

install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md
  DESTINATION ${REVOICE_DATA_DIR} COMPONENT Runtime)
install(FILES libs/pandora/LICENSE
  DESTINATION ${REVOICE_DATA_DIR}/licenses RENAME Pandora-LGPL-2.1.txt COMPONENT Runtime)
include(cmake/Packaging.cmake)
